**24. Based on the three tables below, write a query to get the names of customers who have spent more than the average order amount across all orders.**

```sql
CREATE TABLE Customers (
  customer_id INT PRIMARY KEY,
  customer_name VARCHAR(255) NOT NULL,
  city VARCHAR(100)
);

CREATE TABLE Products (
  product_id INT PRIMARY KEY,
  product_name VARCHAR(255) NOT NULL,
  price DECIMAL(10, 2)
);

CREATE TABLE Orders (
  order_id INT PRIMARY KEY,
  customer_id INT,
  product_id INT,
  total_amount DECIMAL(10, 2),
  FOREIGN KEY (customer_id) REFERENCES Customers(customer_id),
  FOREIGN KEY (product_id) REFERENCES Products(product_id)
);
```

- A. `SELECT c.customer_name FROM Customers c JOIN Orders o ON c.customer_id = o.customer_id WHERE o.total_amount > AVG(o.total_amount);`
- B. `SELECT c.customer_name FROM Customers c JOIN Orders o ON c.customer_id = o.customer_id WHERE o.total_amount > (SELECT AVG(total_amount) FROM Orders);` ✅
- C. `SELECT c.customer_name FROM Customers c WHERE total_amount > (SELECT AVG(total_amount) FROM Orders);`
- D. `SELECT c.customer_name, AVG(o.total_amount) FROM Customers c JOIN Orders o ON c.customer_id = o.customer_id GROUP BY c.customer_name HAVING AVG(o.total_amount) > AVG(o.total_amount);`

**25. Based on the three tables below, display the department names, employee names, and total tasks completed by each employee in the "Engineering" department. Sort the results by total tasks completed in descending order.**

```sql
CREATE TABLE Departments (
  dept_id INT PRIMARY KEY,
  dept_name VARCHAR(255) NOT NULL
);

CREATE TABLE Employees (
  employee_id INT PRIMARY KEY,
  employee_name VARCHAR(255) NOT NULL,
  dept_id INT,
  FOREIGN KEY (dept_id) REFERENCES Departments(dept_id)
);

CREATE TABLE Tasks (
  task_id INT PRIMARY KEY,
  employee_id INT,
  dept_id INT,
  status VARCHAR(50) NOT NULL,
  FOREIGN KEY (employee_id) REFERENCES Employees(employee_id),
  FOREIGN KEY (dept_id) REFERENCES Departments(dept_id)
);
```

- A. `SELECT d.dept_name, e.employee_name, COUNT(t.task_id) AS total_tasks FROM Employees e JOIN Departments d ON e.dept_id = d.dept_id JOIN Tasks t ON e.employee_id = t.employee_id WHERE d.dept_name = 'Engineering' GROUP BY d.dept_name, e.employee_name ORDER BY total_tasks DESC;` ✅
- B. `SELECT d.dept_name, e.employee_name, SUM(t.task_id) AS total_tasks FROM Employees e JOIN Departments d ON e.dept_id = d.dept_id JOIN Tasks t ON e.employee_id = t.employee_id WHERE d.dept_name = 'Engineering' GROUP BY d.dept_name, e.employee_name ORDER BY total_tasks DESC;`
- C. `SELECT d.dept_name, e.employee_name, COUNT(t.task_id) AS total_tasks FROM Employees e JOIN Departments d ON e.dept_id = d.dept_id WHERE d.dept_name = 'Engineering' GROUP BY d.dept_name, e.employee_name ORDER BY total_tasks DESC;`
- D. `SELECT d.dept_name, e.employee_name, COUNT(t.task_id) AS total_tasks FROM Tasks t JOIN Departments d ON t.dept_id = d.dept_id WHERE d.dept_name = 'Engineering' GROUP BY d.dept_name, e.employee_name ORDER BY total_tasks ASC;`
